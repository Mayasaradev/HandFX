cmake_minimum_required(VERSION 3.22.1)
project(HandFX)

add_library(handfx SHARED
    src/main/cpp/main.cpp
    src/main/cpp/hand_tracker.cpp
    src/main/cpp/ui/imgui.cpp
    src/main/cpp/ui/imgui_draw.cpp
    src/main/cpp/ui/imgui_tables.cpp
    src/main/cpp/ui/imgui_impl_opengl3.cpp
    src/main/cpp/ui/imgui_impl_android.cpp
    ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c
)

add_library(tensorflowlite SHARED IMPORTED)
set_target_properties(tensorflowlite PROPERTIES IMPORTED_LOCATION
    ${CMAKE_SOURCE_DIR}/src/main/cpp/libs/${ANDROID_ABI}/libtensorflowlite.so)

target_include_directories(handfx PRIVATE
    src/main/cpp/
    src/main/cpp/libs/include
    ${ANDROID_NDK}/sources/android/native_app_glue
)

find_library(log-lib log)
find_library(android-lib android)
find_library(egl-lib EGL)
find_library(gles-lib GLESv3)
find_library(cam-lib camera2ndk)
find_library(media-lib mediandk)

target_link_libraries(handfx
    tensorflowlite
    ${log-lib} ${android-lib} ${egl-lib} ${gles-lib} ${cam-lib} ${media-lib}
)
